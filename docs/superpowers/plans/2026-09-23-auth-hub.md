## Global Constraints

- Repo: `/Users/gilbertvirgo/vpcc-auth` (new), hosted at `auth.vpcc.church` on Netlify.
- Env: `CALENDAR_MONGODB_URI` (same value as `~/vpcc-calendar/.env` `MONGODB_URI`), `JWT_SECRET` (identical in all three Netlify sites), `COOKIE_DOMAIN` (default `.vpcc.church`).
- Cookie: `vpcc_session=<jwt>; Domain=.vpcc.church; Path=/; HttpOnly; Secure; SameSite=Lax; Max-Age=604800`. When host is not under `COOKIE_DOMAIN`, omit `Domain` and `Secure`.
- JWT payload `{ id, username, role }`, `expiresIn: '7d'`, signed with `JWT_SECRET`, the same shape calendar's `requireAuth` verifies.
- Password check: bcryptjs compare when stored value matches `/^\$2[aby]\$\d{2}\$/`, else plaintext equality. **No migration write.**
- Login response body: `{ user, token }`; 401 `{ error: 'Invalid credentials' }` on failure. No rate limiting.
- CORS: echo `Origin` when its hostname ends with `.vpcc.church` or is `localhost`, with `Access-Control-Allow-Credentials: true`.
- `returnTo` is followed only if its host ends with `.vpcc.church` or is `localhost`; otherwise go to `https://vpcc.church`.
- UI uses vpcc-v1 tokens (copied CSS) and Area Inktrap via Typekit `https://use.typekit.net/ccy7tqi.css`.
- Never print secret values (`JWT_SECRET`, Mongo URI, passwords) in logs or terminal output.
- Every commit message ends with `Co-Authored-By: Claude Fable 5.1 <noreply@anthropic.com>`.

const DEFAULT_RETURN = "https://vpcc.church";

export function returnToIsSafe(url: string): boolean {
  try {
    const u = new URL(url);
    if (u.protocol !== "https:" && u.protocol !== "http:") return false;
    return u.hostname.endsWith(".vpcc.church") || u.hostname === "localhost";
  } catch {
    return false;
  }
}

export function safeReturnTo(url: string | null): string {
  return url && returnToIsSafe(url) ? url : DEFAULT_RETURN;
}

// Copied verbatim into vpcc-sunday-sheets/netlify/functions/_shared/auth.ts. Keep self-contained.
import jwt from "jsonwebtoken";

export type SessionUser = { id: string; username: string; role: string };

const COOKIE = "vpcc_session";
const MAX_AGE = 604800; // 7 days, matches JWT expiresIn

function cookie(value: string, host: string, maxAge: number): string {
  const domain = process.env.COOKIE_DOMAIN || ".vpcc.church";
  const onDomain = host.endsWith(domain);
  const parts = [`${COOKIE}=${value}`];
  if (onDomain) parts.push(`Domain=${domain}`);
  parts.push("Path=/", "HttpOnly");
  if (onDomain) parts.push("Secure");
  parts.push("SameSite=Lax", `Max-Age=${maxAge}`);
  return parts.join("; ");
}

export function sessionCookie(token: string, host: string): string {
  return cookie(token, host, MAX_AGE);
}

export function clearCookie(host: string): string {
  return cookie("", host, 0);
}

export function cors(req: Request): Record<string, string> {
  const origin = req.headers.get("origin");
  if (!origin) return {};
  let host: string;
  try {
    host = new URL(origin).hostname;
  } catch {
    return {};
  }
  if (!host.endsWith(".vpcc.church") && host !== "localhost") return {};
  return {
    "Access-Control-Allow-Origin": origin,
    "Access-Control-Allow-Credentials": "true",
    "Access-Control-Allow-Headers": "Content-Type, Authorization",
    "Access-Control-Allow-Methods": "GET, POST, OPTIONS",
    Vary: "Origin",
  };
}

function readCookie(header: string | null, name: string): string | null {
  for (const part of (header ?? "").split(";")) {
    const [key, ...rest] = part.trim().split("=");
    if (key === name) return rest.join("=");
  }
  return null;
}

export function verifySession(req: Request): SessionUser | null {
  const secret = process.env.JWT_SECRET;
  if (!secret) throw new Error("JWT_SECRET not set");
  const bearer = req.headers.get("authorization")?.match(/^Bearer\s+(.+)$/i)?.[1];
  for (const token of [readCookie(req.headers.get("cookie"), COOKIE), bearer]) {
    if (!token) continue;
    try {
      const p = jwt.verify(token, secret);
      if (typeof p === "string") continue;
      return { id: String(p.id), username: String(p.username), role: String(p.role) };
    } catch {
      // invalid or expired; try the next source
    }
  }
  return null;
}
```

import bcrypt from "bcryptjs";

const BCRYPT = /^\$2[aby]\$\d{2}\$/;

// Mirrors vpcc-calendar User.comparePassword, minus the plaintext->bcrypt migration write.
export async function checkPassword(candidate: string, stored: unknown): Promise<boolean> {
  if (typeof stored !== "string" || !stored) return false;
  if (BCRYPT.test(stored)) return bcrypt.compare(candidate, stored);
  return candidate === stored; // ponytail: legacy plaintext; calendar's own login migrates it
}

import { type Db, MongoClient } from "mongodb";

let client: Promise<MongoClient> | undefined;

// Cached across warm invocations; a failed connect clears the cache so the next call retries.
export async function calendarDb(): Promise<Db> {
  const uri = process.env.CALENDAR_MONGODB_URI;
  if (!uri) throw new Error("CALENDAR_MONGODB_URI not set");
  client ??= new MongoClient(uri, { serverSelectionTimeoutMS: 10000 })
    .connect()
    .catch((err) => {
      client = undefined;
      throw err;
    });
  return (await client).db();
}
```

- [ ] **Step 2: Write `netlify/functions/login.mts`**

```ts
import jwt from "jsonwebtoken";
import { type SessionUser, cors, sessionCookie } from "./_shared/auth";
import { calendarDb } from "./_shared/db";
import { checkPassword } from "./_shared/password";

type UserDoc = { username: string; password: string; role: string };

export default async (req: Request) => {
  const headers = cors(req);
  if (req.method === "OPTIONS") return new Response(null, { status: 204, headers });
  if (req.method !== "POST") {
    return Response.json({ error: "Method not allowed" }, { status: 405, headers });
  }

  const secret = process.env.JWT_SECRET;
  if (!secret) {
    console.error("JWT_SECRET is not set");
    return Response.json({ error: "Server misconfiguration" }, { status: 500, headers });
  }

  const body = await req.json().catch(() => ({}));
  const { username, password } = (body ?? {}) as Record<string, unknown>;
  // Strings only: blocks Mongo operator injection like {"username":{"$ne":null}}.
  if (typeof username !== "string" || typeof password !== "string" || !username || !password) {
    return Response.json({ error: "username and password required" }, { status: 400, headers });
  }

  const db = await calendarDb();
  const user = await db.collection<UserDoc>("users").findOne({ username });
  if (!user || !(await checkPassword(password, user.password))) {
    return Response.json({ error: "Invalid credentials" }, { status: 401, headers });
  }

  const payload: SessionUser = { id: String(user._id), username: user.username, role: user.role };
  const token = jwt.sign(payload, secret, { expiresIn: "7d" });
  const host = new URL(req.url).hostname;

  return Response.json(
    { user: payload, token },
    { headers: { ...headers, "Set-Cookie": sessionCookie(token, host) } },
  );
};
```

- [ ] **Step 4: Write `src/main.ts`**

```ts
import "./style.css";
import { safeReturnTo } from "./returnTo";

const form = document.querySelector<HTMLFormElement>("#login")!;
const error = document.querySelector<HTMLParagraphElement>("#error")!;
const button = form.querySelector<HTMLButtonElement>("button")!;

form.addEventListener("submit", async (e) => {
  e.preventDefault();
  error.textContent = "";
  button.disabled = true;
  const data = new FormData(form);
  try {
    const res = await fetch("/api/login", {
      method: "POST",
      credentials: "include",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ username: data.get("username"), password: data.get("password") }),
    });
    if (res.ok) {
      location.assign(safeReturnTo(new URLSearchParams(location.search).get("returnTo")));
      return;
    }
    error.textContent =
      res.status === 401 ? "Wrong username or password." : "Something went wrong. Try again.";
  } catch {
    error.textContent = "Could not reach the server. Try again.";
  }
  button.disabled = false;
});
```

- [ ] **Step 1: Write `README.md`**

````markdown
# vpcc-auth

Shared login for VPCC tools, served at **https://auth.vpcc.church**.
Signs users in against the calendar's `users` collection and sets a
`vpcc_session` JWT cookie on `.vpcc.church`, accepted by
calendar.vpcc.church and trifold.vpcc.church.

## Endpoints

| Method | Path | Result |
|---|---|---|
| POST | `/api/login` `{username,password}` | 200 `{ user, token }` + `Set-Cookie: vpcc_session=…`; 401 on bad credentials |
| POST | `/api/logout` | Clears the cookie |
| GET | `/api/me` | 200 `{ user }` or 401 |

Page: `/?returnTo=<url>` redirects there after login if the host ends with
`.vpcc.church` or is `localhost`; otherwise to https://vpcc.church.

## Environment variables

| Name | Required | Notes |
|---|---|---|
| `CALENDAR_MONGODB_URI` | yes | Same value as `MONGODB_URI` in the calendar site |
| `JWT_SECRET` | yes | **Must be identical** in the calendar, trifold and auth sites |
| `COOKIE_DOMAIN` | no | Defaults to `.vpcc.church`. On any other host (e.g. localhost) the cookie is set without `Domain` and `Secure` |

## Local development

```bash
npm install
# build .env from the calendar's without printing secrets
grep -E '^(MONGODB_URI|JWT_SECRET)=' ~/vpcc-calendar/.env | sed 's/^MONGODB_URI=/CALENDAR_MONGODB_URI=/' > .env
npx netlify dev        # http://localhost:8888
npm test               # vitest
```

## Netlify setup

1. New site from this repo. Build settings come from `netlify.toml`
   (`npm run build`, publish `dist`, functions `netlify/functions`, Node 24).
2. Site configuration → Environment variables: add `CALENDAR_MONGODB_URI` and
   `JWT_SECRET` (copy from the calendar site; do not generate a new secret).
3. Domain management → add custom domain `auth.vpcc.church`; create the
   `CNAME auth → <site>.netlify.app` DNS record; wait for the HTTPS certificate.
4. Make sure MongoDB Atlas network access allows Netlify (same rule the calendar uses).
````
